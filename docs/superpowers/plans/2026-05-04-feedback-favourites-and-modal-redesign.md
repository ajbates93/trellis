# Feedback Favourites & Modal Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the redundant "Recent Feedback" strip with a curated Favourites feature, and redesign the feedback modal to be theme-neutral.

**Architecture:** Add an `is_favourite` boolean to the `feedback` table. Expose a `PATCH /api/feedback/:id/favourite` endpoint with reviewee-only authorization. Surface the favourite toggle in the feedback table (hover-reveal) and the modal header. Render a Favourites strip at the top of the Received tab, computed client-side from the existing received-feedback response. Replace the dark gradient banner in the modal with a clean theme-neutral header.

**Tech Stack:** Nuxt 4, Vue 3, Drizzle ORM, PostgreSQL, Vitest, Nuxt UI, Tailwind.

**Spec:** `docs/superpowers/specs/2026-05-04-feedback-favourites-and-modal-redesign-design.md`

## Task 5: Favourite toggle endpoint — implementation

**Files:**
- Create: `server/api/feedback/[id]/favourite.patch.ts`

- [ ] **Step 1: Implement the handler**

Create `server/api/feedback/[id]/favourite.patch.ts`:

```ts
import { db } from "#server/database/connection";
import { feedback } from "#server/database/schema";
import { eq } from "drizzle-orm";
import { validateResponse } from "#server/utils/validation";
import { apiSchemas } from "#server/schemas";

export default defineEventHandler(async (event) => {
  const session = event.context.session;

  if (!session?.user) {
    throw createError({
      statusCode: 401,
      message: "Unauthorized - Please log in",
    });
  }

  const feedbackId = getRouterParam(event, "id");

  if (!feedbackId) {
    throw createError({
      statusCode: 400,
      message: "Feedback ID is required",
    });
  }

  const body = await readBody(event);

  if (typeof body?.isFavourite !== "boolean") {
    throw createError({
      statusCode: 400,
      message: "isFavourite must be a boolean",
    });
  }

  try {
    const [feedbackItem] = await db
      .select({
        id: feedback.id,
        revieweeId: feedback.revieweeId,
      })
      .from(feedback)
      .where(eq(feedback.id, feedbackId))
      .limit(1);

    if (!feedbackItem) {
      throw createError({
        statusCode: 404,
        message: "Feedback not found",
      });
    }

    if (feedbackItem.revieweeId !== session.user.id) {
      throw createError({
        statusCode: 403,
        message: "You can only favourite your own feedback",
      });
    }

    const [updatedFeedback] = await db
      .update(feedback)
      .set({ isFavourite: body.isFavourite })
      .where(eq(feedback.id, feedbackId))
      .returning();

    const response = { success: true, feedback: updatedFeedback };
    return validateResponse(response, apiSchemas.SuccessResponseSchema);
  } catch (error) {
    console.error("Error toggling feedback favourite:", error);
    if (error.statusCode) {
      throw error;
    }
    throw createError({
      statusCode: 500,
      message: "Failed to toggle feedback favourite",
    });
  }
});
```

## Task 6: FavouriteToggle component

**Files:**
- Create: `app/components/feedback/FavouriteToggle.vue`

- [ ] **Step 1: Create the component**

Create `app/components/feedback/FavouriteToggle.vue`:

```vue
<script setup lang="ts">
const props = defineProps<{
  feedbackId: string
  isFavourite: boolean
  size?: 'xs' | 'sm' | 'md'
}>()

const emit = defineEmits<{
  'update:isFavourite': [value: boolean]
}>()

const toast = useToast()
const isPending = ref(false)

const toggle = async (e: MouseEvent) => {
  e.stopPropagation()
  if (isPending.value) return

  const next = !props.isFavourite
  // Optimistic update
  emit('update:isFavourite', next)
  isPending.value = true

  try {
    await $api(`/feedback/${props.feedbackId}/favourite`, {
      method: 'PATCH',
      body: { isFavourite: next },
    })
  } catch (error) {
    // Rollback
    emit('update:isFavourite', !next)
    console.error('Failed to toggle favourite:', error)
    toast.add({
      title: 'Failed to update favourite',
      color: 'error',
      icon: 'i-heroicons-x-circle-20-solid',
    })
  } finally {
    isPending.value = false
  }
}
</script>

<template>
  <UButton
    :icon="isFavourite ? 'i-heroicons-star-solid' : 'i-heroicons-star'"
    :color="isFavourite ? 'warning' : 'neutral'"
    variant="ghost"
    :size="size ?? 'sm'"
    :aria-label="isFavourite ? 'Remove from favourites' : 'Add to favourites'"
    :aria-pressed="isFavourite"
    :loading="isPending"
    @click="toggle"
  />
</template>
```

defineProps<{
  favourites: FeedbackWithReviewer[]
}>()

const emit = defineEmits<{
  open: [feedback: FeedbackWithReviewer]
  'unfavourite': [feedbackId: string]
}>()

const formatRelativeTime = (date: Date | string | null | undefined) => {
  if (!date) return ''
  const d = new Date(date)
  const diffMs = Date.now() - d.getTime()
  const days = Math.floor(diffMs / 86_400_000)
  if (days === 0) return 'today'
  if (days === 1) return 'yesterday'
  if (days < 7) return `${days}d ago`
  if (days < 30) return `${Math.floor(days / 7)}w ago`
  if (days < 365) return `${Math.floor(days / 30)}mo ago`
  return `${Math.floor(days / 365)}y ago`
}
</script>

<template>
  <div v-if="favourites.length > 0" class="mb-6">

  modelValue: boolean;
  feedback: (FeedbackWithReviewer | FeedbackWithReviewee) | null;
  showMarkAsRead?: boolean;
}

const props = withDefaults(defineProps<Props>(), {
  showMarkAsRead: false,
});

const emit = defineEmits<{
  "update:modelValue": [value: boolean];
  markAsRead: [feedbackId: string];
  "update:isFavourite": [feedbackId: string, value: boolean];
}>();

const isOpen = computed({
  get: () => props.modelValue,
  set: (value: boolean) => emit("update:modelValue", value),
});

const formatDate = (date: string | Date | undefined | null) => {
  if (!date) return "";
  return format(new Date(date), "MMM d, yyyy");
};

const displayPerson = computed(() => {
  if (!props.feedback) return null;
  if ("reviewer" in props.feedback && props.feedback.reviewer) {
    return { label: "Feedback from", ...props.feedback.reviewer };
  }
  if ("reviewee" in props.feedback && props.feedback.reviewee) {
    return { label: "Feedback to", ...props.feedback.reviewee };
  }
  return null;
});

// Favouriting only applies to feedback the current user received
// (i.e. when displayPerson is the reviewer, not the reviewee).
const canFavourite = computed(() => {
  if (!props.feedback) return false;
  return "reviewer" in props.feedback && props.feedback.reviewer != null;
});

const handleMarkAsRead = () => {
  if (props.feedback) {
    emit("markAsRead", props.feedback.id);
  }
};
</script>
